\subsection{Root pairingから定まるCartan行列}

\begin{definition}
  \label{def:RootPairing.Base}
  \lean{RootPairing.Base}
  \leanok
  $P$をroot pairingとする．
  $b \subseteq \iota$を有限部分集合とする．
  次を満たすとき，$b$は$P$の\textbf{base}であるという：
\begin{enumerate}[label=(\roman*)]
  \item $\{\alpha_i\}_{i \in b}$は$R$上線型独立である．
  \item $\{\alpha^\vee_i\}_{i \in b}$は$R$上線型独立である．
  \item 任意の$i \in \iota$に対し，$\alpha_i$は次のように表せる：
  \begin{equation}
    \alpha_i = \sum_{j \in b} c_j \alpha_j \quad \lor \quad
    -\alpha_i = \sum_{j \in b} c_j \alpha_j \qquad (c_j \in \N).
  \end{equation}
  \item 任意の$i \in \iota$に対し，$\alpha^\vee_i$は次のように表せる：
  \begin{equation}
    \alpha^\vee_i = \sum_{j \in b} c_j \alpha^\vee_j \quad \lor \quad
    -\alpha^\vee_i = \sum_{j \in b} c_j \alpha^\vee_j \qquad (c_j \in \N).
  \end{equation}
\end{enumerate}
\end{definition}

\begin{definition}
  \label{def:RootPairing.Base.cartanMatrix}
  \lean{RootPairing.Base.cartanMatrix}
  \leanok
  $P$はroot pairingでcrystallographicであるとする．
  また，$b$は$P$のbaseとする．
  $C \in \M_b(\Z)$を次のように定め，root pairing $P$から定まる\textbf{Cartan行列}という：
  \begin{equation}
    C_{ij} = B(\alpha_i, \alpha^\vee_j).
  \end{equation}
\end{definition}

\begin{lemma}
  \label{lem:RootPairing.Base.cartanMatrix_mem_of_ne}
  \lean{RootPairing.Base.cartanMatrix_mem_of_ne}
  \leanok
  $P$はroot pairingでcrystallographicであるとする．
  また，$b$は$P$のbaseとする．
  $\iota$は有限で$R$は整域であるとする．
  任意の$i, j \in \iota$に対し，$i \ne j$ならば$C_{ij} \in \{-3, -2, -1, 0\}$である．
\end{lemma}

\begin{lemma}
  \label{lem:RootPairing.Base.cartanMatrix_apply_eq_zero_iff_symm}
  \lean{RootPairing.Base.cartanMatrix_apply_eq_zero_iff_symm}
  \leanok
  \uses{}
  $P$はroot pairingでcrystallographicであるとする．
  また，$b$は$P$のbaseとする．
  $R$は整域であるとする．
  任意の$i, j \in \iota$に対し，$C_{ij} = 0$と$C_{ji} = 0$は同値である．
\end{lemma}
